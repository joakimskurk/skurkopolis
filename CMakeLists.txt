cmake_minimum_required(VERSION 3.16)
project(MetropolisLab VERSION 0.8.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(metro_core core/sequencer.cpp)
target_include_directories(metro_core PUBLIC core)
target_compile_options(metro_core PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
add_executable(metro_trace host/trace.cpp)
target_link_libraries(metro_trace PRIVATE metro_core)
add_executable(core_tests tests/core_test.cpp)
target_link_libraries(core_tests PRIVATE metro_core)

add_library(metro_virtual virtual/synthetic_peripherals.cpp)
target_include_directories(metro_virtual PUBLIC virtual)
target_compile_options(metro_virtual PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
add_executable(peripheral_contract_tests tests/peripheral_contract_test.cpp)
target_link_libraries(peripheral_contract_tests PRIVATE metro_virtual)
enable_testing()
add_test(NAME core COMMAND core_tests)
add_test(NAME peripheral_contracts COMMAND peripheral_contract_tests)
add_executable(response_protocol_tests virtual/response_protocol.c tests/response_protocol_test.c)
target_include_directories(response_protocol_tests PRIVATE virtual)
target_compile_options(response_protocol_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME response_protocols COMMAND response_protocol_tests)
add_custom_target(avr COMMAND sh ${CMAKE_SOURCE_DIR}/scripts/build-avr.sh WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})

add_custom_target(sim COMMAND sh ${CMAKE_SOURCE_DIR}/scripts/build-sim.sh WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_custom_target(inspector COMMAND sh ${CMAKE_SOURCE_DIR}/scripts/build-inspector.sh WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
